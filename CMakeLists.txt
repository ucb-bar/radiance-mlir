cmake_minimum_required(VERSION 3.20)
project(RadianceMLIR LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-fno-rtti)
find_package(MLIR REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
set(MUON_MLIR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../muon-mlir" CACHE PATH "Muon MLIR source")
set(MX_GEMMINI_MLIR_SOURCE_DIR "" CACHE PATH "MX Gemmini MLIR source")
if(NOT EXISTS "${MUON_MLIR_SOURCE_DIR}/include/Muon/MuonOps.td")
  message(FATAL_ERROR "Set MUON_MLIR_SOURCE_DIR to the muon-mlir source tree")
endif()
if(NOT EXISTS "${MX_GEMMINI_MLIR_SOURCE_DIR}/include/MxGemmini/MxOps.td")
  message(FATAL_ERROR "Set MX_GEMMINI_MLIR_SOURCE_DIR to the mx-gemmini-mlir source tree")
endif()
add_subdirectory(${MUON_MLIR_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/muon EXCLUDE_FROM_ALL)
add_subdirectory(${MX_GEMMINI_MLIR_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/mx EXCLUDE_FROM_ALL)
include_directories(${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS}
  ${MUON_MLIR_SOURCE_DIR}/include ${MX_GEMMINI_MLIR_SOURCE_DIR}/include
  ${CMAKE_CURRENT_BINARY_DIR}/muon/include ${CMAKE_CURRENT_BINARY_DIR}/mx/include
  ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_definitions(${LLVM_DEFINITIONS})
add_subdirectory(lib)
add_subdirectory(tools)
